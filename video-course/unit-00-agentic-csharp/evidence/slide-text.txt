Slide 32
MCP Interactive Input
MRTR – Multi-Round-Trip Requests
•
The MCP server may need more input before it
can complete the request
▪
Return Input Required
▪
The client obtains the missing input from the user or
application
▪
MRTR enables interaction and replaces old Elicitation,
Sampling & Roots calls.
[McpServerTool]
public async Task<string> RestoreScheduledModeAsync(
McpServer server,
RequestContext<CallToolRequestParams> context,
string assetId,
CancellationToken cancellationToken) {
if (TryGetApproval(context, out bool approved))
return approved ?
await RestoreAsync(assetId, cancellationToken) : "Cancelled";
if (!server.IsMrtrSupported)
return "Interactive input not supported";
throw new InputRequiredException(
inputRequests: new Dictionary<string, InputRequest> {
["approval"] = InputRequest.ForElicitation(
new ElicitRequestParams
{
Message =
$"Restore {assetId} to scheduled mode?",
RequestedSchema = approvalSchema
})
});
}
private static bool TryGetApproval(
RequestContext<CallToolRequestParams> context,
out bool approved)
{
approved = false;
if (context.Params?.InputResponses?.TryGetValue(
"approval", out var response) != true)
{
return false;
}
var result = response.Deserialize(
InputResponse.ElicitResultJsonTypeInfo);
approved = result?.Action == "accept";
return true;
}
